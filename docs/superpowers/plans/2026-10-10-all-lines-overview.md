# All-lines overview Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "Trim the Fat" button in the game bar's mod area opens a window listing every player line with its verdict, balance per year, average load and vehicle count, worst first, with F2-style transport filters and a summary line.

**Architecture:** The line card's line-level calculation moves into `line_data.lineInput`, which the card and the window share, so their verdicts agree. Pure additions to `metrics.tl` (verdict kind, severity, summary) are unit-tested. Text helpers move from `line_card.script.tl` into a shared `ui.tl`. The window is a tool registered with `tool_react_util.registerToolWithWindow` and pushed on the game's default tool stack, as the base Music Player does.

**Tech Stack:** Teal 0.24.8 against the TF3 tealdefs (`just check`), plain-Lua tests (`just test`), TF3 React-like UI (`builtin.*`, `react.*`).

**Spec:** `docs/superpowers/specs/2026-10-10-all-lines-overview-design.md`. Kit notes: `../AGENTS.md` (read "Adding UI to entity windows", "Translations", "Teal gotchas").

**Hard rules (crash the whole game UI if broken):**
- Every recipe returns a layout (`builtin.BoxLayout{...}`), never a bare `TextView`/`ImageView`, including DataTable cell recipes.
- Engine calls in UI code run inside `pcall`.
- DataTable cells keep their first `userParam`: put mutable state in a table the cells read, and change `meta.localKey` to remount the table when rows or filters change.

**Paths** (all under `mod/pgrace_trim_the_fat/`):
- Canonical types: `tealdef/gui/pgrace_trim_the_fat/*.d.tl`. The copies under `content/` are generated by `just check`; never edit them.
- Module registry: `tealdef/pgrace_trim_the_fat_def.d.tl`.
- Code: `content/gui/pgrace_trim_the_fat/`.

```teal
function metrics.verdictKind(a : PgraceTrimTheFatMetrics.Assessment) : PgraceTrimTheFatMetrics.VerdictKind
	if a.verdict == "Losing" then
		if a.notDelivering then
			return "LosingStuck"
		elseif a.trim > 0 then
			return "LosingTrim"
		end
		return "Losing"
	end
	return a.verdict as PgraceTrimTheFatMetrics.VerdictKind
end

local SEVERITY : {PgraceTrimTheFatMetrics.VerdictKind : integer} = {
	Losing = 1, LosingTrim = 1, LosingStuck = 1,
	TooMany = 2, Busy = 3, TooNew = 4, NoVehicles = 5, Ok = 6,
}

function metrics.severity(kind : PgraceTrimTheFatMetrics.VerdictKind) : integer
	return SEVERITY[kind] or 6
end

function metrics.summarise(assessments : {PgraceTrimTheFatMetrics.Assessment}) : PgraceTrimTheFatMetrics.Summary
	local s : PgraceTrimTheFatMetrics.Summary = { losingCount = 0, losingPerYear = 0, tooManyCount = 0, trimTotal = 0 }
	for __, a in ipairs(assessments) do
		if a.verdict == "Losing" then
			s.losingCount = s.losingCount + 1
			s.losingPerYear = s.losingPerYear + (a.balancePerYear or 0)
		end
		if a.trim > 0 then
			s.tooManyCount = s.tooManyCount + 1
			s.trimTotal = s.trimTotal + a.trim
		end
	end
	return s
end
```

```teal
-- The line-level numbers, and the vehicle rows when withRows is set (the overview skips them).
local function collect(line : Engine.Entity, withRows : boolean) : PgraceTrimTheFatLineData.LineFacts, {VehicleRow}
	local now = nowMs()
	local yearMs = api.util.getDefaultYearDuration()
	local from = math.max(now - yearMs, 0)
	local rows : {VehicleRow} = {}
	local count = 0
	local oldest : integer = nil
	local vehicleDataMs = 0
	local carrier : Carrier = nil
	for __, vehicle in ipairs(api.engine.system.transportVehicleSystem.getLineVehicles(line)) do
		local tv = transportVehicle(vehicle)
		if tv ~= nil then
			count = count + 1
			local bought = purchaseTime(tv) or now
			if oldest == nil or bought < oldest then
				oldest = bought
			end
			if carrier == nil then
				carrier = tv.carrier
			end
			local spanMs = dataSpan({ vehicle }, bought, now, yearMs)
			vehicleDataMs = vehicleDataMs + spanMs
			if withRows then
				local balance = api.engine.util.finance.calculateBalance({ vehicle }, from, now, true)
				local perYear, annualised = metrics.annualise(balance, spanMs, yearMs)
				rows[#rows + 1] = {
					entity = vehicle,
					name = api.engine.util.getEntityName(vehicle),
					ageText = api.engine.util.formatAge(bought, now),
					ageMs = now - bought,
					balancePerYear = perYear,
					annualised = annualised,
					loadNow = line_data.vehicleLoadNow(vehicle),
				}
			end
		end
	end
	local __, capacity, cargoTypes, passengers = capacityTotals(line, false)
	local frequency = api.engine.util.line.getMaxFrequency(line)
	local lineComponent = api.engine.getComponent(line, api.type.ComponentType.LINE) as Engine.Component.Line
	local facts : PgraceTrimTheFatLineData.LineFacts = {
		input = {
			vehicleCount = count,
			balance = api.engine.util.finance.calculateBalance({ line }, from, now, true),
			itemsTransported = api.engine.util.logbook.getLogValuePerYear(line, "itemsTransported"),
			totalCapacity = capacity,
			-- getMaxFrequency is departures per second; the line window shows 1 / frequency seconds.
			intervalMs = (frequency ~= nil and frequency > 0.0001) and 1000 / frequency or nil,
			stopCount = lineComponent ~= nil and #lineComponent.stops or nil,
			dataMs = oldest ~= nil and dataSpan({ line }, oldest, now, yearMs) or 0,
			vehicleDataMs = vehicleDataMs,
			yearMs = yearMs,
			cargoTypeCount = cargoTypes,
			loadNow = line_data.lineLoadNow(line),
			carriesPassengers = passengers,
		},
		carrier = carrier,
		noun = nounFor(carrier, passengers),
	}
	return facts, rows
end

function line_data.lineInput(line : Engine.Entity) : PgraceTrimTheFatLineData.LineFacts
	local facts = collect(line, false)
	return facts
end

function line_data.snapshot(line : Engine.Entity) : PgraceTrimTheFatLineData.LineSnapshot
	local facts, rows = collect(line, true)
	return {
		input = facts.input,
		vehicles = rows,
		signature = line_data.vehicleSignature(line),
		noun = facts.noun,
	}
end

function line_data.playerLines() : {Engine.Entity}
	return api.engine.system.lineSystem.getLinesForPlayer(api.engine.util.getPlayer())
end
```

-- PROBE (remove before release): how long assessing every line takes on the player's save.
local function probeAllLines() : string
	local start = os.clock()
	local lines = line_data.playerLines()
	local failed = 0
	for __, line in ipairs(lines) do
		local ok = pcall(function() : boolean
			metrics.assess(line_data.lineInput(line).input)
			return true
		end)
		if not ok then
			failed = failed + 1
		end
	end
	local ms = math.floor((os.clock() - start) * 1000 + 0.5)
	local text = tostring(#lines) .. " lines, " .. tostring(failed) .. " failed, " .. tostring(ms) .. " ms"
	debugPrint(LOG_PREFIX .. "probe overview " .. text)
	return text
end

local record Row
	line : Engine.Entity
	name : string
	error : boolean
	assessment : Assessment
	vehicleCount : integer
	-- matches[i]: the line belongs to categories()[i].
	matches : {integer : boolean}
	verdictText : string
	verdictClass : string
	-- Severity, then balance per year (lower first); errors last.
	verdictSort : number
	balanceText : string
	loadText : string
	vehiclesText : string

local record OverviewData
	error : boolean
	rows : {integer : Row}
	keys : {integer}
	-- The player's sort, kept across refreshes. Starts on Per year (column 3), lowest first.
	sortColumn : integer
	sortAscending : boolean
	-- Differs per build, so a refresh remounts the table (cells keep their first userParam).
	generation : integer

local function verdictText(kind : VerdictKind, a : Assessment) : string, string
	local trim = ui.formatCount(a.trim)
	if kind == "LosingStuck" then
		return _("Losing · not delivering"), "font-scale-body, negative"
	elseif kind == "LosingTrim" then
		return ui.fill(_("Losing money (−{count})"), { count = trim }), "font-scale-body, negative"
	elseif kind == "Losing" then
		return _("Losing money"), "font-scale-body, negative"
	elseif kind == "TooMany" then
		return ui.fill(_("Too many (−{count})"), { count = trim }), "font-scale-body, warning"
	elseif kind == "Busy" then
		return _("Busy"), "font-scale-body, warning"
	elseif kind == "TooNew" then
		return _("Too new"), "font-scale-body"
	elseif kind == "NoVehicles" then
		return _("No vehicles"), "font-scale-body"
	end
	return _("OK"), "font-scale-body, positive"
end

local ERROR_SEVERITY = 7
-- Keeps severity groups apart in one number: balances stay far below this.
local SEVERITY_SCALE = 1e13

local function buildRow(line : Engine.Entity, cats : {Category}) : Row
	local row : Row = { line = line, matches = {}, error = true }
	local nameOk, name = pcall(api.engine.util.getEntityName, line)
	row.name = nameOk and name as string or tostring(line)
	for i, cat in ipairs(cats) do
		local ok, match = pcall(line_util.filterLine, { cat.carrier }, line)
		row.matches[i] = ok and match as boolean
	end
	local ok, err = pcall(function() : boolean
		local facts = line_data.lineInput(line)
		local a = metrics.assess(facts.input)
		local kind = metrics.verdictKind(a)
		row.assessment = a
		row.vehicleCount = facts.input.vehicleCount
		row.verdictText, row.verdictClass = verdictText(kind, a)
		row.verdictSort = metrics.severity(kind) * SEVERITY_SCALE + (a.balancePerYear or 0)
		row.balanceText = ui.moneyText(a.balancePerYear, a.annualised)
		row.loadText = ui.formatPercent(a.load)
		row.vehiclesText = ui.vehiclesText(facts.noun, facts.input.vehicleCount)
		row.error = false
		return true
	end)
	if not ok then
		debugPrint(LOG_PREFIX .. "overview error for line " .. tostring(line) .. ": " .. tostring(err))
		row.verdictText, row.verdictClass = _("Error"), "font-scale-body, negative"
		row.verdictSort = ERROR_SEVERITY * SEVERITY_SCALE
		row.balanceText, row.loadText, row.vehiclesText = ui.UNKNOWN, ui.UNKNOWN, ui.UNKNOWN
	end
	return row
end

local nextGeneration = 0

-- Never throws.
local function buildOverview(cats : {Category}) : OverviewData
	nextGeneration = nextGeneration + 1
	local data : OverviewData = { error = false, rows = {}, keys = {}, sortColumn = 3, sortAscending = true,
		generation = nextGeneration }
	local start = os.clock() -- PROBE (remove before release)
	local ok, err = pcall(function() : boolean
		for __, line in ipairs(line_data.playerLines()) do
			data.rows[line] = buildRow(line, cats)
			data.keys[#data.keys + 1] = line
		end
		return true
	end)
	debugPrint(LOG_PREFIX .. "probe overview " .. tostring(#data.keys) .. " lines in " -- PROBE (remove before release)
		.. tostring(math.floor((os.clock() - start) * 1000 + 0.5)) .. " ms") -- PROBE (remove before release)
	if not ok then
		debugPrint(LOG_PREFIX .. "overview error: " .. tostring(err))
		data.error = true
	end
	return data
end
