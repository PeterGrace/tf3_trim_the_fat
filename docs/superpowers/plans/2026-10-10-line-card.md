# Line Card Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a localised "Line efficiency" card to the TF3 line window. It shows a verdict (Losing / Too many / Busy / OK), a balance and load summary on a 12-month basis, and a per-vehicle table.

**Architecture:** There are three modules:
- `metrics.tl` is pure: verdict maths, constants and plural rules, unit-tested in plain Lua.
- `line_data.tl` is the only file that calls `api.engine`. It builds a plain `LineSnapshot`.
- `line_card.script.tl` is the React UI, registered on `LineEowExtensionPoint`.

All visible text goes through `_()` with translations in `strings.json` for the 12 TF3 languages.

**Tech Stack:** Teal 0.24.8 (type-checked against the game tealdefs), TF3 react UI (`builtin`, `content_card`, `DataTable`), plain-Lua test runner (`tools/test.lua`), `just`.

**Spec:** `docs/superpowers/specs/2026-10-10-line-card-design.md`

  `tools/build.py` copies them next to the `.tl` files, so never edit the copies under `content/`.
- **Commands:**
  - `just check`: build plus type-check;
  - `just test`: unit tests;
  - `just verify`: check, test and release-check (which includes the translations);
  - `just install`: symlinks the mod into TF3's staging area.
- **The agent cannot run the game.** In-game verification is Task 8, done by the user.
- **Reference implementation:** `../tf3_commuter_compass/mod/pgrace_commuter_compass/content/gui/pgrace_commuter_compass/pgrace_commuter_compass.script.tl`
  (`countText`, `currentLanguage`, `formatCount`, `DataTable` usage).
- **Engine facts used below** (from the base game source):
  - **Balance:** the base Balance is `api.engine.util.finance.calculateBalance({entity}, now - yearMs, now, true)`.
    That's the last 12 months of income minus running and maintenance costs, without the purchase price.
  - **Game time:** `GAME_TIME.gameTime` and `api.util.getDefaultYearDuration()` are in milliseconds.
  - **Frequency:** `api.engine.util.line.getMaxFrequency(line)` is departures per second. The line window shows
    `1 / frequency` as its interval in seconds. Probe 2 confirms the units.
  - **Capacity:** `api.engine.util.line.getLineCapacityUsages(line, false)` maps `cargoTypeId + 1` to `{used, capacity}`
    for the current load configuration.
  - **Transported:** `api.engine.util.logbook.getLogValuePerYear(line, "itemsTransported")` is the line's
    "Transported" figure for the last year.
  - **Purchase time:** `TransportVehiclePart.purchaseTime`, per part. A vehicle's age is that of its oldest part
    (as in `vehicle_util.getAge`). `api.engine.util.formatAge(t0, t1)` gives a localised age string.

```teal
-- Pure helpers (no api.* calls), so tests/metrics_test.lua can run them in plain Lua.
global record PgraceTrimTheFatMetrics
	enum Verdict
		"NoVehicles"
		"TooNew"
		"Losing"
		"TooMany"
		"Busy"
		"Ok"
	end

	-- What line_data.tl reads from the engine for one line. Times are game time in milliseconds.
	record LineInput
		vehicleCount : integer
		-- Money over the last 12 months (less for a young line): income minus running and maintenance costs,
		-- purchase price excluded, as the base game's Balance.
		balance : integer
		-- Items carried over the last 12 months, all cargo types (the line window's "Transported").
		itemsTransported : integer
		-- Sum of the vehicles' capacities in their current load configuration, all cargo types.
		totalCapacity : number
		-- Time between departures at a stop; nil or <= 0 when unknown.
		intervalMs : number
		-- Length of the data window: min(one year, time since the oldest vehicle was bought).
		dataMs : number
		yearMs : number
		-- Cargo types the vehicles currently carry (capacity > 0).
		cargoTypeCount : integer
	end

	record Assessment
		verdict : Verdict
		-- Average load over the data window, nil when unknown (no capacity, interval or data).
		load : number
		target : number
		-- Balance scaled to a full year; equal to the balance once the window is a full year.
		balancePerYear : number
		-- True when balancePerYear was scaled up from less than a year.
		annualised : boolean
		-- balancePerYear / vehicleCount; nil when there are no vehicles.
		profitPerVehicle : number
		-- Vehicles to keep and to remove. trim is 0 (and keep nil) when no trim is suggested.
		keep : integer
		trim : integer
	end

	-- Target average load for lines carrying several cargo types or passengers.
	TARGET_DEFAULT : number
	-- Target for single-cargo lines, which usually run empty one way.
	TARGET_SINGLE_CARGO : number
	-- At or above this average load the verdict is Busy.
	BUSY_LOAD : number
	-- Below this share of a year of data the verdict is TooNew.
	MIN_DATA_FRACTION : number

	-- TARGET_SINGLE_CARGO for exactly one cargo type, TARGET_DEFAULT otherwise.
	targetFor : function(cargoTypeCount : integer) : number
	-- itemsTransported / (departures in the window × capacity per vehicle); nil when unknown.
	averageLoad : function(input : LineInput) : number
	-- amount × yearMs / spanMs and true when 0 < spanMs < yearMs; otherwise amount and false.
	annualise : function(amount : number, spanMs : number, yearMs : number) : number, boolean
	-- The verdict and the numbers behind it. See the spec §1 for the rules.
	assess : function(input : LineInput) : Assessment
	-- Whole percent, rounded half up (0.344 → 34); nil for nil.
	percent : function(fraction : number) : integer
	-- The plural form a language uses for a count: "one", "few", "many" or "other" (CLDR integer rules).
	-- The language is a TF3 code such as "de" or "pt_BR"; unknown ones get English's one/other.
	pluralCategory : function(language : string, count : integer) : string
end
```

```teal
local metrics : PgraceTrimTheFatMetrics = {}

metrics.TARGET_DEFAULT = 0.70
metrics.TARGET_SINGLE_CARGO = 0.50
metrics.BUSY_LOAD = 0.95
metrics.MIN_DATA_FRACTION = 1 / 12

function metrics.targetFor(cargoTypeCount : integer) : number
	if cargoTypeCount == 1 then
		return metrics.TARGET_SINGLE_CARGO
	end
	return metrics.TARGET_DEFAULT
end

function metrics.percent(fraction : number) : integer
	if fraction == nil then
		return nil
	end
	return math.floor(fraction * 100 + 0.5)
end

-- Integer plural rules from CLDR, for the languages TF3 ships (same as Commuter Compass rank.pluralCategory).
local function slavicCategory(count : integer, oneNeedsTensCheck : boolean) : string
	local lastDigit, lastTwo = count % 10, count % 100
	if count == 1 or (oneNeedsTensCheck and lastDigit == 1 and lastTwo ~= 11) then
		return "one"
	end
	if lastDigit >= 2 and lastDigit <= 4 and not (lastTwo >= 12 and lastTwo <= 14) then
		return "few"
	end
	return "many"
end

function metrics.pluralCategory(language : string, count : integer) : string
	local lang = (language or "en"):sub(1, 2)
	if lang == "ja" or lang == "ko" or lang == "zh" then
		return "other"
	end
	if lang == "ru" then
		return slavicCategory(count, true)
	end
	if lang == "pl" then
		return slavicCategory(count, false)
	end
	if lang == "fr" or lang == "pt" then
		return (count == 0 or count == 1) and "one" or "other"
	end
	return count == 1 and "one" or "other"
end

return metrics
```

```teal
function metrics.averageLoad(input : PgraceTrimTheFatMetrics.LineInput) : number
	if input.vehicleCount == nil or input.vehicleCount <= 0
		or input.totalCapacity == nil or input.totalCapacity <= 0
		or input.intervalMs == nil or input.intervalMs <= 0
		or input.dataMs == nil or input.dataMs <= 0 then
		return nil
	end
	local departures = input.dataMs / input.intervalMs
	local offered = departures * (input.totalCapacity / input.vehicleCount)
	return (input.itemsTransported or 0) / offered
end

function metrics.annualise(amount : number, spanMs : number, yearMs : number) : number, boolean
	if spanMs == nil or spanMs <= 0 or spanMs >= yearMs then
		return amount, false
	end
	return amount * yearMs / spanMs, true
end
```

```teal
-- Guards ceil() against float noise: 8 × 0.25 ÷ 0.5 may come out as 4.000000001.
local EPSILON = 1e-9

function metrics.assess(input : PgraceTrimTheFatMetrics.LineInput) : PgraceTrimTheFatMetrics.Assessment
	local a : PgraceTrimTheFatMetrics.Assessment = {
		verdict = "Ok",
		target = metrics.targetFor(input.cargoTypeCount),
		trim = 0,
	}
	a.balancePerYear, a.annualised = metrics.annualise(input.balance or 0, input.dataMs, input.yearMs)
	local count = input.vehicleCount or 0
	if count <= 0 then
		a.verdict = "NoVehicles"
		return a
	end
	a.profitPerVehicle = a.balancePerYear / count
	a.load = metrics.averageLoad(input)
	if input.dataMs == nil or input.dataMs < input.yearMs * metrics.MIN_DATA_FRACTION then
		a.verdict = "TooNew"
		return a
	end
	if a.load ~= nil and a.load < a.target and count > 1 then
		local keep = math.max(1, math.ceil(count * a.load / a.target - EPSILON))
		if keep < count then
			a.keep = keep
			a.trim = count - keep
		end
	end
	if a.balancePerYear < 0 then
		a.verdict = "Losing"
	elseif a.trim > 0 then
		a.verdict = "TooMany"
	elseif a.load ~= nil and a.load >= metrics.BUSY_LOAD then
		a.verdict = "Busy"
	end
	return a
end
```

```teal
local metrics = ug_require "pgrace_trim_the_fat::/gui/pgrace_trim_the_fat/metrics.tl" as PgraceTrimTheFatMetrics

local line_data : PgraceTrimTheFatLineData = {}

local LOG_PREFIX = "[trim_the_fat] "

local type VehicleRow = PgraceTrimTheFatLineData.VehicleRow
local type Noun = PgraceTrimTheFatLineData.Noun

local function nowMs() : integer
	local gameTime = api.engine.getComponent(api.engine.util.getWorld(), api.type.ComponentType.GAME_TIME) as Engine.Component.GameTime
	return gameTime.gameTime
end

local function transportVehicle(vehicle : Engine.Entity) : Engine.Component.TransportVehicle
	return api.engine.getComponent(vehicle, api.type.ComponentType.TRANSPORT_VEHICLE) as Engine.Component.TransportVehicle
end

-- A vehicle is as old as its oldest part (as the base game's vehicle_util.getAge).
local function purchaseTime(tv : Engine.Component.TransportVehicle) : integer
	local oldest : integer = nil
	for __, part in ipairs(tv.transportVehicleConfig.vehicles) do
		if oldest == nil or part.purchaseTime < oldest then
			oldest = part.purchaseTime
		end
	end
	return oldest
end

local function nounFor(carrier : Carrier, carriesPassengers : boolean) : Noun
	local carriers = api.type["enum"].Carrier
	if carrier == carriers.RAIL then
		return "train"
	elseif carrier == carriers.TRAM then
		return "tram"
	elseif carrier == carriers.WATER then
		return "ship"
	elseif carrier == carriers.AIR then
		return "aircraft"
	elseif carrier == carriers.ROAD then
		return carriesPassengers and "bus" or "truck"
	end
	return "vehicle"
end

-- Used and total capacity over the cargo types the vehicles currently carry, how many such types, and whether
-- passengers are among them.
local function capacityTotals(line : Engine.Entity) : number, number, integer, boolean
	local usages = api.engine.util.line.getLineCapacityUsages(line, false)
	local passengerIndex = api.res.cargoTypeRep.getPassengerCargoTypeId() + 1
	local used, capacity, types, passengers = 0.0, 0.0, 0, false
	for index, usage in pairs(usages or {}) do
		if usage.capacity > 0 then
			used = used + usage.used
			capacity = capacity + usage.capacity
			types = types + 1
			if index == passengerIndex then
				passengers = true
			end
		end
	end
	return used, capacity, types, passengers
end

function line_data.lineLoadNow(line : Engine.Entity) : number
	local used, capacity = capacityTotals(line)
	if capacity <= 0 then
		return nil
	end
	return used / capacity
end

function line_data.vehicleLoadNow(vehicle : Engine.Entity) : number
	local tv = transportVehicle(vehicle)
	local used, capacity = 0, 0
	-- capacities[i] is the capacity for cargo type i - 1 in the current load configuration.
	for i, cap in ipairs(tv.config.capacities) do
		if cap > 0 then
			capacity = capacity + cap
			used = used + api.engine.system.simEntityAtVehicleSystem.getVehicleSimEntitiesCountForCargoType(vehicle, i - 1)
		end
	end
	if capacity <= 0 then
		return nil
	end
	return used / capacity
end

function line_data.snapshot(line : Engine.Entity) : PgraceTrimTheFatLineData.LineSnapshot
	local now = nowMs()
	local yearMs = api.util.getDefaultYearDuration()
	local from = math.max(now - yearMs, 0)
	local rows : {VehicleRow} = {}
	local oldest : integer = nil
	local carrier : Carrier = nil
	for __, vehicle in ipairs(api.engine.system.transportVehicleSystem.getLineVehicles(line)) do
		local tv = transportVehicle(vehicle)
		local bought = purchaseTime(tv) or now
		if oldest == nil or bought < oldest then
			oldest = bought
		end
		if carrier == nil then
			carrier = tv.carrier
		end
		local ageMs = now - bought
		local balance = api.engine.util.finance.calculateBalance({ vehicle }, from, now, true)
		local perYear, annualised = metrics.annualise(balance, ageMs, yearMs)
		rows[#rows + 1] = {
			entity = vehicle,
			name = api.engine.util.getEntityName(vehicle),
			ageText = api.engine.util.formatAge(bought, now),
			ageMs = ageMs,
			balancePerYear = perYear,
			annualised = annualised,
			loadNow = line_data.vehicleLoadNow(vehicle),
		}
	end
	local __, capacity, cargoTypes, passengers = capacityTotals(line)
	local frequency = api.engine.util.line.getMaxFrequency(line)
	return {
		input = {
			vehicleCount = #rows,
			balance = api.engine.util.finance.calculateBalance({ line }, from, now, true),
			itemsTransported = api.engine.util.logbook.getLogValuePerYear(line, "itemsTransported"),
			totalCapacity = capacity,
			-- getMaxFrequency is departures per second; the line window shows 1 / frequency seconds.
			intervalMs = (frequency ~= nil and frequency > 0.0001) and 1000 / frequency or nil,
			dataMs = oldest ~= nil and math.min(yearMs, now - oldest) or 0,
			yearMs = yearMs,
			cargoTypeCount = cargoTypes,
		},
		vehicles = rows,
		noun = nounFor(carrier, passengers),
	}
end

-- PROBE (remove before release): the raw inputs, to check units and calibrate the targets in-game.
function line_data.probe(line : Engine.Entity, snap : PgraceTrimTheFatLineData.LineSnapshot, assessment : PgraceTrimTheFatMetrics.Assessment)
	local ok, err = pcall(function() : boolean
		local i = snap.input
		debugPrint(LOG_PREFIX .. "probe line " .. tostring(line)
			.. " name=" .. tostring(api.engine.util.getEntityName(line))
			.. " frequency=" .. tostring(api.engine.util.line.getMaxFrequency(line))
			.. " intervalMs=" .. tostring(i.intervalMs)
			.. " vehicles=" .. tostring(i.vehicleCount)
			.. " capacity=" .. tostring(i.totalCapacity)
			.. " cargoTypes=" .. tostring(i.cargoTypeCount)
			.. " items=" .. tostring(i.itemsTransported)
			.. " dataMs=" .. tostring(i.dataMs)
			.. " yearMs=" .. tostring(i.yearMs)
			.. " balance=" .. tostring(i.balance)
			.. " loadA=" .. tostring(assessment.load)
			.. " loadNow=" .. tostring(line_data.lineLoadNow(line))
			.. " verdict=" .. tostring(assessment.verdict)
			.. " noun=" .. tostring(snap.noun))
		for __, row in ipairs(snap.vehicles) do
			local logOk, items = pcall(api.engine.util.logbook.getLogValuePerYear, row.entity, "itemsTransported")
			debugPrint(LOG_PREFIX .. "probe vehicle " .. tostring(row.entity)
				.. " itemsTransportedOk=" .. tostring(logOk)
				.. " itemsTransported=" .. tostring(items)
				.. " ageMs=" .. tostring(row.ageMs)
				.. " balancePerYear=" .. tostring(row.balancePerYear))
		end
		return true
	end)
	if not ok then
		debugPrint(LOG_PREFIX .. "probe failed: " .. tostring(err))
	end
end

return line_data
```

```teal
-- Text helpers ---------------------------------------------------------------------------------------------------

-- Replaces {name} placeholders; unknown names are left as they are.
local function fill(template : string, values : {string : string}) : string
	return (template:gsub("{(%w+)}", function(key : string) : string
		return values[key] or ("{" .. key .. "}")
	end))
end

local function currentLanguage() : string
	local ok, code = pcall(function() : string
		return app.getUserProfile():getLanguage().code
	end)
	return ok and code as string or "en"
end

local function formatCount(count : integer) : string
	local ok, text = pcall(api.util.lang.formatInt, count)
	if ok and text ~= nil then
		return text as string
	end
	return tostring(count)
end

local function formatMoney(amount : number) : string
	local rounded = math.floor(amount + 0.5)
	local ok, text = pcall(api.util.formatMoney, rounded)
	if ok and text ~= nil then
		return text as string
	end
	return tostring(rounded)
end

-- Money with "*" when it was scaled up from less than a year (see the table footnote).
local function moneyText(amount : number, annualised : boolean) : string
	return formatMoney(amount) .. (annualised and "*" or "")
end

-- "34%", or as the player's language writes it (German "34 %"); UNKNOWN for nil.
local function formatPercent(fraction : number) : string
	local whole = metrics.percent(fraction)
	if whole == nil then
		return UNKNOWN
	end
	return fill(_("{percent}%"), { percent = tostring(whole) })
end

-- "5.4", with the language's decimal separator.
local function formatOneDecimal(value : number) : string
	local tenths = math.floor(value * 10 + 0.5)
	return fill(_("{integer}.{decimal}"), {
		integer = string.format("%d", math.floor(tenths / 10)),
		decimal = string.format("%d", math.floor(tenths % 10)),
	})
end

-- A count with its noun, in the form the player's language uses for that count. Each form is its own key; mods get
-- no plural-aware gettext, so the rule is ours (metrics.pluralCategory). Copied from Commuter Compass.
local record PluralForms
	one : string
	few : string
	many : string
	other : string
end

local function countText(count : integer, forms : PluralForms) : string
	local category = metrics.pluralCategory(currentLanguage(), count)
	local template = (forms as {string : string})[category] or forms.other
	return fill(template, { count = formatCount(count) })
end

-- Built once when the window opens and handed to the table as userParam (cells keep their first userParam).
local record CardData
	error : boolean
	snapshot : LineSnapshot
	assessment : Assessment
	-- Vehicle rows by entity, which are also the table's row keys.
	rows : {integer : VehicleRow}
	-- The latest polled load per vehicle, written by the Load cells.
	liveLoad : {integer : number}
	-- The copy the table sorts by, refreshed on a header click, so live updates don't reorder rows (AGENTS.md).
	sortLoad : {integer : number}

local function buildCardData(line : Engine.Entity) : CardData
	local ok, snapOrError = pcall(line_data.snapshot, line)
	if not ok then
		debugPrint(LOG_PREFIX .. "error for line " .. tostring(line) .. ": " .. tostring(snapOrError))
		return { error = true }
	end
	local snap = snapOrError as LineSnapshot
	local data : CardData = {
		error = false,
		snapshot = snap,
		assessment = metrics.assess(snap.input),
		rows = {},
		liveLoad = {},
		sortLoad = {},
	}
	for __, row in ipairs(snap.vehicles) do
		data.rows[row.entity] = row
		data.liveLoad[row.entity] = row.loadNow
		data.sortLoad[row.entity] = row.loadNow or -1
	end
	line_data.probe(line, snap, data.assessment) -- PROBE (remove before release)
	return data
end

-- Verdict -------------------------------------------------------------------------------------------------------

local function verdictLabel(verdict : PgraceTrimTheFatMetrics.Verdict) : string, string
	if verdict == "NoVehicles" then
		return _("No vehicles on this line"), "font-scale-body"
	elseif verdict == "TooNew" then
		return _("Too new to judge"), "font-scale-body"
	elseif verdict == "Losing" then
		return _("Losing money"), "font-scale-body, negative"
	elseif verdict == "TooMany" then
		return _("Too many vehicles"), "font-scale-body, warning"
	elseif verdict == "Busy" then
		return _("Busy: may need more vehicles"), "font-scale-body, warning"
	end
	return _("OK"), "font-scale-body, positive"
end

local function verdictTooltip(data : CardData) : string
	local a = data.assessment
	local count = data.snapshot.input.vehicleCount
	local parts : {string} = {}
	if a.verdict == "TooNew" then
		parts[#parts + 1] = _("This line has less than a month of data.")
	end
	if a.verdict == "Losing" then
		parts[#parts + 1] = _("The balance over the last 12 months is negative: income minus running and maintenance costs, without the purchase price.")
	end
	if a.trim > 0 then
		parts[#parts + 1] = fill(_("{count} × {load} ÷ {target} = {exact}, so keep {keep}."), {
			count = formatCount(count),
			load = formatPercent(a.load),
			target = formatPercent(a.target),
			exact = formatOneDecimal(count * a.load / a.target),
			keep = formatCount(a.keep),
		})
		parts[#parts + 1] = _("Fewer vehicles means a longer wait, which can cost passengers or cargo.")
	end
	if a.verdict == "Busy" then
		parts[#parts + 1] = fill(_("The average load is {busy} or more, so vehicles may leave passengers or cargo behind."),
			{ busy = formatPercent(metrics.BUSY_LOAD) })
	end
	if a.verdict == "Ok" then
		parts[#parts + 1] = _("Profitable, and the average load is close to or above the target.")
	end
	return table.concat(parts, "\n")
end

-- Starts sorted by balance per year, lowest first: the vehicles to look at first are on top.
local function makeVehicleTable(data : CardData) : TreeNodeId
	local rowKeys : {integer} = {}
	for i, row in ipairs(data.snapshot.vehicles) do
		rowKeys[i] = row.entity
	end
	return builtin.DataTable{
		columns = {
			builtin.ColumnDesc{
				name = _("Vehicle"),
				recipe = NameCell,
				weight = 1,
				getCompareValue = function(rowKey : integer) : any
					return data.rows[rowKey].name
				end,
			},
			builtin.ColumnDesc{
				name = _("Age"),
				recipe = AgeCell,
				getCompareValue = function(rowKey : integer) : any
					return data.rows[rowKey].ageMs
				end,
			},
			builtin.ColumnDesc{
				name = _("Balance per year"),
				recipe = BalanceCell,
				getCompareValue = function(rowKey : integer) : any
					return data.rows[rowKey].balancePerYear
				end,
			},
			builtin.ColumnDesc{
				name = _("Load now"),
				recipe = LoadCell,
				getCompareValue = function(rowKey : integer) : any
					return data.sortLoad[rowKey] or -1
				end,
			},
		},
		rowKeys = rowKeys,
		userParam = data,
		initialSortColumn = { 3, true },
		onSortColumnChange = function(_column : integer, _ascending : boolean)
			local copy : {integer : number} = {}
			for key, load in pairs(data.liveLoad) do
				copy[key] = load
			end
			data.sortLoad = copy
		end,
	}
```
